from pathlib import Path
from urllib.parse import urlparse  # , urljoin
import httpx
import sys
import time
from bs4 import BeautifulSoup

OUT = Path.home() / "images" / time.strftime("%Y%m%d/%H%M%S")

with httpx.Client(follow_redirects=True, timeout=30) as client:
    with (
        Path("img-src.txt").open("a") as f_src,
        Path("a-href.txt").open("a") as f_href,
    ):
        for i, u in enumerate(sys.stdin, 1):
            u = u.strip()
            if not u:
                continue

            for attempt in range(6):
                try:
                    with client.stream("GET", u) as response:
                        response.raise_for_status()

                        content_type = response.headers.get("content-type", "").lower()

                        if content_type.startswith("image/"):
                            OUT.mkdir(parents=True, exist_ok=True)
                            filename = (
                                OUT
                                / f"image_{i:02d}{Path(urlparse(u).path).suffix or '.jpg'}"
                            )
                            print(f"IMG: {u} ({content_type})", file=sys.stderr)
                            with filename.open("wb") as f:
                                for chunk in response.iter_bytes(
                                    chunk_size=1024 * 1024
                                ):
                                    f.write(chunk)
                            continue

                        if content_type.startswith("text/"):
                            print(f"TEXT: {u} ({content_type})", file=sys.stderr)
                            response.read()
                            soup = BeautifulSoup(response.text, "lxml")

                            for img_index, img in enumerate(
                                soup.select("img"), start=1
                            ):  # data-srcset
                                src = img.get("data-lazy-src") or img.get("src")
                                if not src:
                                    continue

                                f_src.write(src + "\n")

                            for img_index, a in enumerate(soup.select("a"), start=1):
                                href = a.get("href")
                                if not href:
                                    continue

                                f_href.write(href + "\n")

                                # Resolve the href relative to base URL of current page
                                # full_url = urljoin(u, href)

                            continue

                        if attempt == 5:
                            response.raise_for_status()

                        retry_after = response.headers.get("Retry-After")
                        delay = (
                            float(retry_after)
                            if retry_after and retry_after.isdigit()
                            else 2**attempt
                        )
                        print(
                            f"{response.status_code}: retrying {u} in {delay}s",
                            file=sys.stderr,
                        )
                        time.sleep(delay)

                except (
                    httpx.ConnectError,
                    httpx.ReadError,
                    httpx.TimeoutException,
                ) as e:
                    if attempt == 5:
                        print(f"FAILED: {u}: {e}", file=sys.stderr)
                        break

                    delay = 2**attempt
                    print(
                        f"{type(e).__name__}: retrying {u} in {delay}s", file=sys.stderr
                    )
                    time.sleep(delay)
